-- 1. REMOTE CACHE & ENGINE OPTIMIZATION
----------------------------------------------------------------------
local RemoteCache = {}
local function cacheRemotes()
    for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
        if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
            RemoteCache[string.lower(obj.Name)] = obj
        end
    end
end
cacheRemotes()

local function getRemote(nameList)
    for _, name in ipairs(nameList) do
        local found = RemoteCache[string.lower(name)]
        if found then return found end
    end
    return nil
end

local function fireRemote(nameList, ...)
    local remote = getRemote(nameList)
    if remote then
        if remote:IsA("RemoteEvent") then
            remote:FireServer(...)
        elseif remote:IsA("RemoteFunction") then
            remote:InvokeServer(...)
        end
    end
end

local function safeTeleport(cframeTarget)
    local char = localPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root and cframeTarget then
        root.CFrame = cframeTarget + Vector3.new(0, 3, 0)
    end
end

-- 3. UNIFIED HIGH-PERFORMANCE BACKGROUND ENGINE
----------------------------------------------------------------------
local attackDebounce = tick()
local shopDebounce = tick()

task.spawn(function()
    while true do
        task.wait(Config.TPDelay)
        local char = localPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")

        if root then
            -- 1. Auto Attack & Mob Farming
            if Config.AutoFarmSelectedMob or Config.AutoFarmNearestMob or Config.AutoClick then
                local searchContainers = {
                    Workspace:FindFirstChild("Enemies"),
                    Workspace:FindFirstChild("Mobs"),
                    Workspace:FindFirstChild("Bosses"),
                    Workspace
                }

                local targetMob = nil
                for _, container in ipairs(searchContainers) do
                    if container then
                        for _, mob in ipairs(container:GetChildren()) do
                            local mobRoot = mob:FindFirstChild("HumanoidRootPart")
                            local mobHum = mob:FindFirstChild("Humanoid")
                            if mobRoot and mobHum and mobHum.Health > 0 and mob ~= char then
                                if Config.AutoFarmSelectedMob and mob.Name == Config.SelectedMob then
                                    targetMob = mob
                                    break
                                elseif Config.AutoFarmNearestMob then
                                    targetMob = mob
                                    break
                                end
                            end
                        end
                    end
                    if targetMob then break end
                end

                if targetMob and targetMob:FindFirstChild("HumanoidRootPart") then
                    local mobPos = targetMob.HumanoidRootPart.CFrame
                    if Config.FarmMethod == "TP" then
                        root.CFrame = mobPos + Vector3.new(0, 2, 2)
                    elseif Config.FarmMethod == "Tween" then
                        local tweenInfo = TweenInfo.new(Config.TweenDelay, Enum.EasingStyle.Linear)
                        TweenService:Create(root, tweenInfo, {CFrame = mobPos + Vector3.new(0, 2, 2)}):Play()
                    end

                    -- Check Dynamic Team Switch on Mob HP%
                    if Config.AutoChangeTeam then
                        local mobHum = targetMob:FindFirstChild("Humanoid")
                        if mobHum and mobHum.MaxHealth > 0 then
                            local hpPercent = (mobHum.Health / mobHum.MaxHealth) * 100
                            if hpPercent <= Config.MobPercentHPToTeam then
                                fireRemote({"equipteam", "switchteam", "changeteam"}, Config.SwitchToTeam)
                            else
                                fireRemote({"equipteam", "switchteam", "changeteam"}, 1)
                            end
                        end
                    end

                    fireRemote({"attack", "hit", "punch", "damage"}, targetMob)
                    local tool = char:FindFirstChildOfClass("Tool")
                    if tool then tool:Activate() end
                end
            end

            -- 2. Auto Arise Mobs
            if Config.AutoAriseMobs then
                local searchCorpseFolders = {Workspace:FindFirstChild("Corpses"), Workspace:FindFirstChild("Debris"), Workspace}
                for _, folder in ipairs(searchCorpseFolders) do
                    if folder then
                        for _, corpse in ipairs(folder:GetChildren()) do
                            if corpse:FindFirstChild("SoulPrompt") or string.find(string.lower(corpse.Name), "soul") then
                                fireRemote({"arise", "extract", "extractsoul"}, corpse)
                            end
                        end
                    end
                end
            end

            -- 3. Auto Enchant
            if Config.AutoEnchant then
                fireRemote({"enchant", "autoenchant", "enchantweapon"}, Config.EnchantMethod, Config.TargetEnchantsCount)
            end

            -- 4. Shop Purchases (Throttled Every 2s)
            if tick() - shopDebounce > 2.0 then
                shopDebounce = tick()
                if Config.BuyRarePowderDust then fireRemote({"buyshopitem", "buyitem"}, "RarePowderDust") end
                if Config.BuyLegToRareDust then fireRemote({"buyshopitem", "buyitem"}, "LegToRareDust") end
                if Config.BuyLegendaryDust then fireRemote({"buyshopitem", "buyitem"}, "LegendaryPowderDust") end
                if Config.BuyUltimateRankupRune then fireRemote({"buyshopitem", "buyitem"}, "UltimateRankUpRune") end
                if Config.BuyDoubleDungeonRune then fireRemote({"buyshopitem", "buyitem"}, "DoubleDungeonRune") end
                if Config.AutoBuyWinterShop then fireRemote({"buywintershop", "wintersell"}) end
                if Config.AutoBuyCores then fireRemote({"buycores", "buycore"}) end
                if Config.AutoCraftKeys then fireRemote({"craftkeys", "craftkey"}) end
                if Config.AutoSellShadows then fireRemote({"sellshadows", "sellshadow"}) end
                if Config.AutoSellWeapons then fireRemote({"sellweapons", "sellweapon"}) end
            end

            -- 5. Auto Candies Collect
            if Config.AutoCollectCandies then
                local candyFolder = Workspace:FindFirstChild("Candies") or Workspace:FindFirstChild("Halloween")
                if candyFolder then
                    for _, candy in ipairs(candyFolder:GetChildren()) do
                        if candy:IsA("BasePart") then
                            root.CFrame = candy.CFrame
                            task.wait(0.08)
                        end
                    end
                end
            end
        end
    end
end)
