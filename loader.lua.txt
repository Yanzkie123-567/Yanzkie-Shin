local function humanDelay(minWait, maxWait)
	if not Config.HumanizeEnabled then
		task.wait(minWait or 0.1)
		return
	end
	local rMin = math.floor((minWait or Config.MinPreDelay) * 1000)
	local rMax = math.floor((maxWait or Config.MaxPreDelay) * 1000)
	if rMax <= rMin then rMax = rMin + 50 end
	task.wait(math.random(rMin, rMax) / 1000)
end

local function getActivePetMount()
	local char = localPlayer.Character
	if not char then return nil, nil end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum and hum.SeatPart then
		local model = hum.SeatPart:FindFirstAncestorOfClass("Model")
		if model and model ~= char then
			return model, hum.SeatPart
		end
	end
	for _, obj in ipairs(workspace:GetChildren()) do
		if obj:IsA("Model") and string.find(string.lower(obj.Name), string.lower(localPlayer.Name)) and obj ~= char then
			local primary = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
			if primary then return obj, primary end
		end
	end
	return nil, nil
end

local function getObjectPosition(obj)
	if not obj or not obj.Parent then return nil end
	if obj:IsA("Model") then
		if obj.PrimaryPart then return obj.PrimaryPart.Position end
		local ok, cf = pcall(function() return obj:GetPivot() end)
		if ok and cf then return cf.Position end
	elseif obj:IsA("BasePart") then
		return obj.Position
	elseif obj:IsA("Folder") then
		local part = obj:FindFirstChildWhichIsA("BasePart", true)
		if part then return part.Position end
		if obj.Parent and obj.Parent:IsA("Model") then
			local ok, cf = pcall(function() return obj.Parent:GetPivot() end)
			if ok and cf then return cf.Position end
		end
	end
	return nil
end

local function pointInRotatedBox(point, boxCFrame, boxSize)
	local relative = boxCFrame:PointToObjectSpace(point)
	local half = boxSize * 0.5
	return math.abs(relative.X) <= half.X
		and math.abs(relative.Y) <= half.Y
		and math.abs(relative.Z) <= half.Z
end

local function resolvePlotPosition(obj)
	if not obj then return nil, nil end

	local targetModel = nil
	if obj:IsA("Model") then
		targetModel = obj
	elseif obj.Parent and obj.Parent:IsA("Model") then
		targetModel = obj.Parent
	end

	if targetModel then
		local ok, cf = pcall(function() return targetModel:GetPivot() end)
		if ok and cf then return targetModel, cf.Position end
		if targetModel.PrimaryPart then return targetModel, targetModel.PrimaryPart.Position end
	end

	local fallbackPart = obj:FindFirstChildWhichIsA("BasePart", true) 
		or (targetModel and targetModel:FindFirstChildWhichIsA("BasePart", true))
	if fallbackPart then
		return targetModel or fallbackPart, fallbackPart.Position
	end

	return nil, nil
end

local function autoDetectMyPlot()
	local myName = string.lower(localPlayer.Name)
	local myUserId = tostring(localPlayer.UserId)
	local myDisplayName = string.lower(localPlayer.DisplayName)

	local function isOwnerMatch(val)
		if not val then return false end
		if val == localPlayer then return true end
		local s = string.lower(tostring(val))
		return s == myName or s == myUserId or s == myDisplayName
	end

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Model") or obj:IsA("Folder") then
			local matched = false

			local ownerVal = obj:FindFirstChild("Owner") 
				or obj:FindFirstChild("OwnerName") 
				or obj:FindFirstChild("Player") 
				or obj:FindFirstChild("ClaimedBy") 
				or obj:FindFirstChild("UserId")

			if ownerVal and isOwnerMatch(ownerVal.Value) then matched = true end

			if not matched then
				local attrOwner = obj:GetAttribute("Owner") 
					or obj:GetAttribute("OwnerName") 
					or obj:GetAttribute("Player") 
					or obj:GetAttribute("UserId")
				if attrOwner and isOwnerMatch(attrOwner) then matched = true end
			end

			if matched then
				local model, pos = resolvePlotPosition(obj)
				if pos then
					State.HomePlotModel = model
					State.HomePlotPosition = pos
					return State.HomePlotPosition
				end
			end
		end
	end

	local _, hrp = getCharacterParts()
	if hrp then
		State.HomePlotPosition = hrp.Position
		return State.HomePlotPosition
	end

	State.HomePlotPosition = Vector3.new(0, 10, 0)
	return State.HomePlotPosition
end

local function getMyPlotPosition()
	return State.HomePlotPosition or autoDetectMyPlot()
end

local function isEggOnPlot(obj)
	if not obj then return true end
	local eggPos = getObjectPosition(obj)
	local plotPos = getMyPlotPosition()

	if eggPos and plotPos and (eggPos - plotPos).Magnitude <= Config.PlotRadius then
		return true
	end

	if Config.CheckBoundingBox and State.HomePlotModel and State.HomePlotModel:IsA("Model") and eggPos then
		local success, cf, size = pcall(function() return State.HomePlotModel:GetBoundingBox() end)
		if success and cf and size then
			if pointInRotatedBox(eggPos, cf, size + Vector3.new(8, 8, 8)) then
				return true
			end
		end
	end
	return false
end

local function isEggBlacklisted(eggObj)
	local expireTime = State.FailedEggCache[eggObj]
	if not expireTime then return false end
	if os.clock() < expireTime then return true end
	State.FailedEggCache[eggObj] = nil
	return false
end

local function setNoClip(enable)
	if enable then
		if State.Connections.NoClip then return end
		local stepEvent = RunService.PreSimulation or RunService.Stepped

		State.Connections.NoClip = stepEvent:Connect(function()
			local char, hrp, hum = getCharacterParts()
			if not char or not hrp then return end

			for _, part in ipairs(char:GetDescendants()) do
				if part:IsA("BasePart") and part.CanCollide then
					part.CanCollide = false
				end
			end

			local petModel = getActivePetMount()
			if petModel then
				for _, part in ipairs(petModel:GetDescendants()) do
					if part:IsA("BasePart") and part.CanCollide then
						part.CanCollide = false
					end
				end
			end

			if hum then
				hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
				hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
			end

			if not State.IsRunning and not State.IsCandyRunning and not State.CurrentTween then
				if math.abs(hrp.AssemblyLinearVelocity.Y) < 0.5 then
					hrp.AssemblyLinearVelocity = Vector3.new(hrp.AssemblyLinearVelocity.X, 0, hrp.AssemblyLinearVelocity.Z)
				end
			end
		end)
	else
		if State.Connections.NoClip then
			State.Connections.NoClip:Disconnect()
			State.Connections.NoClip = nil
		end

		local char, hrp, hum = getCharacterParts()
		if char then
			for _, part in ipairs(char:GetDescendants()) do
				if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
					part.CanCollide = true
				end
			end
			if hum then
				hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
				hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
			end
		end
		if hrp then
			hrp.AssemblyLinearVelocity = Vector3.zero
		end
	end
end

local function executeRawTween(targetCFrame, speed)
	local _, hrp, hum = getCharacterParts()
	if not hrp or not hum or hum.Health <= 0 then return nil, 0 end

	if State.CurrentTween then
		pcall(function() State.CurrentTween:Cancel() end)
		State.CurrentTween = nil
	end

	local distance = (targetCFrame.Position - hrp.Position).Magnitude
	local safeSpeed = math.max(15, speed or Config.OutboundSpeed)
	local duration = math.max(distance / safeSpeed, 0.05)

	hrp.AssemblyLinearVelocity = Vector3.zero
	hrp.AssemblyAngularVelocity = Vector3.zero
	if Config.AnchorDuringFlight then
		hrp.Anchored = true
	end

	local _, seatPart = getActivePetMount()
	if seatPart and Config.AnchorDuringFlight then
		pcall(function() seatPart.Anchored = true end)
	end

	local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
	State.CurrentTween = TweenService:Create(hrp, tweenInfo, {CFrame = targetCFrame})
	State.CurrentTween:Play()
	return State.CurrentTween, duration
end

local function safeWaitTween(tween, expectedDuration, targetPosition)
	if not tween then return false end
	local finished = false
	local conn = tween.Completed:Connect(function() finished = true end)
	local startTime = os.clock()

	while not finished and (os.clock() - startTime < (expectedDuration or 5) + 4) do
		if not State.IsRunning and not State.IsCandyRunning then
			pcall(function() tween:Cancel() end)
			if conn then conn:Disconnect() end
			local _, hrp = getCharacterParts()
			if hrp then hrp.Anchored = false end
			local _, seat = getActivePetMount()
			if seat then pcall(function() seat.Anchored = false end) end
			return false
		end

		if targetPosition then
			local _, hrp = getCharacterParts()
			if hrp and (hrp.Position - targetPosition).Magnitude <= 5 then
				finished = true
				break
			end
		end

		task.wait(0.03)
	end
	if conn then conn:Disconnect() end
	return finished
end

local function skyCruiseToPosition(targetPosition, travelSpeed, landOffset)
	local _, hrp = getCharacterParts()
	if not hrp then return false end

	local highAltitudeY = math.max(hrp.Position.Y, targetPosition.Y) + Config.CruiseAltitude

	local climbPos = Vector3.new(hrp.Position.X, highAltitudeY, hrp.Position.Z)
	local tw1, d1 = executeRawTween(CFrame.new(climbPos), travelSpeed * 1.25)
	if not safeWaitTween(tw1, d1, climbPos) or (not State.IsRunning and not State.IsCandyRunning) then return false end

	local cruisePos = Vector3.new(targetPosition.X, highAltitudeY, targetPosition.Z)
	local tw2, d2 = executeRawTween(CFrame.new(cruisePos), travelSpeed)
	if not safeWaitTween(tw2, d2, cruisePos) or (not State.IsRunning and not State.IsCandyRunning) then return false end

	local offset = landOffset or Vector3.new(0, 2, 0)
	local landPos = targetPosition + offset
	local tw3, d3 = executeRawTween(CFrame.new(landPos), travelSpeed * 1.1)
	return safeWaitTween(tw3, d3, landPos)
end

local function smartMoveTo(targetPosition, customSpeed, landOffset)
	local success = false
	local offset = landOffset or Vector3.new(0, 2, 0)
	if Config.MovementMode == "Sky Cruise (Anti-Obstacle)" then
		success = skyCruiseToPosition(targetPosition, customSpeed or Config.OutboundSpeed, offset)
	else
		local destPos = targetPosition + offset
		local tw, expectedTime = executeRawTween(CFrame.new(destPos), customSpeed or Config.OutboundSpeed)
		success = safeWaitTween(tw, expectedTime, destPos)
	end

	local _, hrp = getCharacterParts()
	if hrp then
		hrp.Anchored = false
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
	end
	local _, seat = getActivePetMount()
	if seat then
		pcall(function() seat.Anchored = false end)
	end

	return success
end

local function executeSafeInteraction(targetObj)
	if (not State.IsRunning and not State.IsCandyRunning) or not targetObj or not targetObj.Parent then return false end
	local char, hrp = getCharacterParts()
	if not char or not hrp then return false end

	humanDelay(Config.MinPreDelay, Config.MaxPreDelay)
	local interacted = false

	if targetObj:IsA("BasePart") then
		pcall(function()
			targetObj.CanTouch = true
			targetObj.CanCollide = false
		end)
	end
	for _, d in ipairs(targetObj:GetDescendants()) do
		if d:IsA("BasePart") then
			pcall(function()
				d.CanTouch = true
				d.CanCollide = false
			end)
		end
	end

	local prompts = {}
	if targetObj:IsA("ProximityPrompt") then table.insert(prompts, targetObj) end
	for _, p in ipairs(targetObj:GetDescendants()) do
		if p:IsA("ProximityPrompt") then table.insert(prompts, p) end
	end
	if targetObj.Parent and targetObj.Parent ~= workspace then
		for _, p in ipairs(targetObj.Parent:GetChildren()) do
			if p:IsA("ProximityPrompt") then table.insert(prompts, p) end
		end
	end

	for _, prompt in ipairs(prompts) do
		pcall(function()
			prompt.RequiresLineOfSight = false
			prompt.HoldDuration = 0
			prompt.MaxActivationDistance = 999
			prompt.Enabled = true
		end)
		if fireproximityprompt then
			fireproximityprompt(prompt)
			interacted = true
		end
	end

	local touchingParts = {hrp}
	for _, limbName in ipairs({"LeftFoot", "RightFoot", "Torso", "UpperTorso", "LeftHand", "RightHand"}) do
		local limb = char:FindFirstChild(limbName)
		if limb and limb:IsA("BasePart") then table.insert(touchingParts, limb) end
	end

	local petModel = getActivePetMount()
	if petModel then
		for _, part in ipairs(petModel:GetDescendants()) do
			if part:IsA("BasePart") then table.insert(touchingParts, part) end
		end
	end

	local targetParts = {}
	if targetObj:IsA("BasePart") then table.insert(targetParts, targetObj) end
	for _, d in ipairs(targetObj:GetDescendants()) do
		if d:IsA("BasePart") then table.insert(targetParts, d) end
	end

	for _, tPart in ipairs(targetParts) do
		if firetouchinterest then
			for _, myPart in ipairs(touchingParts) do
				firetouchinterest(myPart, tPart, 0)
				task.wait(0.01)
				firetouchinterest(myPart, tPart, 1)
			end
			interacted = true
		end
	end

	local targetPos = getObjectPosition(targetObj)
	if targetPos then
		hrp.CFrame = CFrame.new(targetPos + Vector3.new(0, 0.5, 0))
		hrp.AssemblyLinearVelocity = Vector3.new(0.5, 0, 0.5)
		task.wait(0.12)
		hrp.AssemblyLinearVelocity = Vector3.zero
	end

	humanDelay(Config.WaitAtEgg, Config.WaitAtEgg + 0.1)
	return interacted
end

local function getEggRarity(obj)
	if not obj then return "unknown" end

	local attr = obj:GetAttribute("Rarity") or obj:GetAttribute("EggRarity") or obj:GetAttribute("Tier")
	if attr and type(attr) == "string" and attr ~= "" then return string.lower(attr) end

	local valObj = obj:FindFirstChild("Rarity") or obj:FindFirstChild("EggRarity") or obj:FindFirstChild("Tier")
	if valObj and valObj:IsA("StringValue") and valObj.Value ~= "" then return string.lower(valObj.Value) end

	local name = string.lower(obj.Name)

	if string.find(name, "blackhole") or string.find(name, "black hole")
		or string.find(name, "solaris") or string.find(name, "cherub")
		or string.find(name, "volcanic") or string.find(name, "ethereal") then
		return "ethereal"
	end

	if string.find(name, "aurora") or string.find(name, "galaxy")
		or string.find(name, "bloom") or string.find(name, "divine") then
		return "divine"
	end

	if string.find(name, "diamond") or string.find(name, "crystal") or string.find(name, "skull")
		or string.find(name, "asteroid") or string.find(name, "dominus") or string.find(name, "flaming")
		or string.find(name, "sinister") or string.find(name, "soul") or string.find(name, "tidal")
		or string.find(name, "mythic") then
		return "mythic"
	end

	if string.find(name, "glass") or string.find(name, "golden") or string.find(name, "gold")
		or string.find(name, "legendary") then
		return "legendary"
	end

	if string.find(name, "mushroom") or string.find(name, "flower") or string.find(name, "slime")
		or string.find(name, "ice") or string.find(name, "epic") then
		return "epic"
	end

	if string.find(name, "cracked") or string.find(name, "easter") or string.find(name, "stone")
		or string.find(name, "leaf") or string.find(name, "rare") then
		return "rare"
	end

	if string.find(name, "white") or string.find(name, "brown") or string.find(name, "common") then
		return "common"
	end

	return "unknown"
end

local function getRootEggObject(obj)
	if not obj then return nil end
	if obj.Parent and obj.Parent ~= workspace and obj.Parent:IsA("Model") then
		local pName = string.lower(obj.Parent.Name)
		if string.find(pName, "egg") or getEggRarity(obj.Parent) ~= "unknown" then
			return obj.Parent
		end
	end
	return obj
end

local function getActiveEggObjects()
	local targetEggs = {}
	local processedSet = {}
	local targetRarity = string.lower(Config.TargetRarity)

	local function processCandidate(candidate)
		if not candidate or not candidate.Parent then return end

		local eggObj = getRootEggObject(candidate)
		if processedSet[eggObj] then return end

		if Players:GetPlayerFromCharacter(eggObj) or (localPlayer.Character and eggObj:IsDescendantOf(localPlayer.Character)) then return end
		if isEggBlacklisted(eggObj) or isEggOnPlot(eggObj) then return end

		local eggName = string.lower(eggObj.Name)
		local detectedRarity = getEggRarity(eggObj)

		local isAnEgg = string.find(eggName, "egg") or (detectedRarity ~= "unknown")
		if not isAnEgg then return end

		local matches = (targetRarity == "all") or (detectedRarity == targetRarity)
		if matches then
			processedSet[eggObj] = true
			table.insert(targetEggs, eggObj)
		end
	end

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Model") or obj:IsA("BasePart") then
			processCandidate(obj)
		end
	end

	return targetEggs
end

-- 10. STRICT "CANDYDROPOFF" SCANNER (EXCLUDES DOORCANDYDROP)
--------------------------------------------------------------------------------
local function getCandyDropOffPoint()
	-- Priority 1: User-saved manual location
	if State.CandyDropOffPosition then
		return State.CandyDropOffInstance, State.CandyDropOffPosition
	end

	-- Priority 2: Direct lookup for exact part named "CandyDropOff" in Workspace
	local exactDrop = workspace:FindFirstChild("CandyDropOff", true)
	if exactDrop then
		local pos = getObjectPosition(exactDrop)
		if pos then
			State.CandyDropOffInstance = exactDrop
			State.CandyDropOffPosition = pos
			return exactDrop, pos
		end
	end

	-- Priority 3: Scan workspace for CandyDropOff (explicitly rejecting DoorCandyDrop)
	local dropKeywords = {"candydropoff", "candy drop-off", "candy drop off"}
	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Model") or obj:IsA("BasePart") then
			local lowerName = string.lower(obj.Name)
			-- Strict filter: NEVER match DoorCandyDrop or anything containing 'door'
			if not string.find(lowerName, "door") and not string.find(lowerName, "spawn") and not string.find(lowerName, "candy_") then
				for _, keyword in ipairs(dropKeywords) do
					if string.find(lowerName, keyword) then
						local pos = getObjectPosition(obj)
						if pos then
							State.CandyDropOffInstance = obj
							State.CandyDropOffPosition = pos
							return obj, pos
						end
					end
				end
			end
		end
	end

	-- Priority 4: Fallback to home plot if CandyDropOff part is not found
	local homePos = getMyPlotPosition()
	return nil, homePos
end

local function getCandySourceContainers()
	local containers = {}

	local hwCandy = workspace:FindFirstChild("HalloweenCandy")
	if hwCandy then table.insert(containers, hwCandy) end

	local hwArea = workspace:FindFirstChild("HalloweenArea")
	if hwArea then
		local candySpawns = hwArea:FindFirstChild("CandySpawns") or hwArea:FindFirstChild("Candy") or hwArea:FindFirstChild("HalloweenCandy")
		if candySpawns then table.insert(containers, candySpawns) end
	end

	if #containers == 0 then
		for _, folder in ipairs(workspace:GetDescendants()) do
			local n = string.lower(folder.Name)
			if (n == "halloweencandy" or n == "candyspawns") and (folder:IsA("Folder") or folder:IsA("Model")) then
				table.insert(containers, folder)
			end
		end
	end

	return containers
end

local function getActiveCandyObjects()
	local containers = getCandySourceContainers()
	local validCandies = {}
	local seen = {}

	for _, container in ipairs(containers) do
		for _, item in ipairs(container:GetChildren()) do
			if (item:IsA("Model") or item:IsA("BasePart")) and item.Parent and not seen[item] then
				if not isEggBlacklisted(item) then
					seen[item] = true
					table.insert(validCandies, item)
				end
			end
		end

		for _, item in ipairs(container:GetDescendants()) do
			if (item:IsA("Model") or item:IsA("BasePart")) and item.Parent and not seen[item] then
				local lower = string.lower(item.Name)
				if string.find(lower, "candy") and not isEggBlacklisted(item) then
					seen[item] = true
					table.insert(validCandies, item)
				end
			end
		end
	end

	return validCandies
end

local function performCandyDeepScan()
	local containers = getCandySourceContainers()
	if #containers == 0 then
		return "No HalloweenCandy folder found in Workspace!"
	end

	local candies = getActiveCandyObjects()
	if #candies == 0 then
		return "Found folder: " .. containers[1]:GetFullName() .. " but no candies currently spawned."
	end

	local sample = candies[1]
	local report = {}
	table.insert(report, "Name: " .. sample.Name)
	table.insert(report, "Class: " .. sample.ClassName)
	table.insert(report, "Path: " .. sample:GetFullName())

	local hasPrompt = false
	local hasTouch = false
	local hasClick = false

	for _, d in ipairs(sample:GetDescendants()) do
		if d:IsA("ProximityPrompt") then hasPrompt = true end
		if d:IsA("TouchTransmitter") then hasTouch = true end
		if d:IsA("ClickDetector") then hasClick = true end
	end

	table.insert(report, "\n[Trigger Systems]")
	table.insert(report, "• ProximityPrompt: " .. (hasPrompt and "DETECTED" or "None"))
	table.insert(report, "• TouchTransmitter: " .. (hasTouch and "DETECTED" or "None"))
	table.insert(report, "• ClickDetector: " .. (hasClick and "DETECTED" or "None"))

	local petModel = getActivePetMount()
	table.insert(report, "\n[Mount Status]")
	table.insert(report, "• Riding Pet: " .. (petModel and ("YES (" .. petModel.Name .. ")") or "NO (On Foot)"))

	local dropObj, dropPos = getCandyDropOffPoint()
	table.insert(report, "\n[CandyDropOff Status (Strict)]")
	if dropObj and dropObj.Name == "CandyDropOff" then
		table.insert(report, "• Target: EXACT CandyDropOff Found! (" .. dropObj:GetFullName() .. ")")
	elseif dropObj then
		table.insert(report, "• Target: " .. dropObj:GetFullName())
	elseif State.CandyDropOffPosition then
		table.insert(report, "• Target: Manual Coordinates Active")
	else
		table.insert(report, "• Target: CandyDropOff part not detected, using Home Plot")
	end

	return table.concat(report, "\n")
end

local function applyPerformanceBoost(enable)
	if not enable then return end
	task.spawn(function()
		pcall(function()
			settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
			local terrain = workspace:FindFirstChildOfClass("Terrain")

				elseif obj:IsA("Beam") or obj:IsA("Highlight") then
					obj.Enabled = false
				end
			end
		end)
	end)
end

local function cleanupAll()
	State.IsRunning = false
	State.IsCandyRunning = false
	if State.CurrentTween then
		pcall(function() State.CurrentTween:Cancel() end)
		State.CurrentTween = nil
	end
	setNoClip(false)
	disableScreenShake(false)

	local _, hrp = getCharacterParts()
	if hrp then
		hrp.Anchored = false
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
	end
	local _, seat = getActivePetMount()
	if seat then pcall(function() seat.Anchored = false end) end

	updateStatus("Idle")
	updateCandyStatus("Idle")
	task.defer(function() pcall(function() collectgarbage("step", 50) end) end)
end

local function serverHop()
	local placeId = game.PlaceId
	local curJob = game.JobId
	local url = "https://games.roblox.com/v1/games/" .. placeId .. "/servers/Public?sortOrder=Asc&limit=100"

	local success, result = pcall(function()
		return HttpService:JSONDecode(game:HttpGet(url))
	end)

	if success and result and result.data then
		for _, server in ipairs(result.data) do
			if server.playing < server.maxPlayers and server.id ~= curJob then
				TeleportService:TeleportToPlaceInstance(placeId, server.id, localPlayer)
				return
			end
		end
	end
	TeleportService:Teleport(placeId, localPlayer)
end

-- 14. MOBILE & DESKTOP FLOATING TOGGLE

task.spawn(function()
	pcall(function()
		local old = CoreGui:FindFirstChild("MobileFluentToggle")
		if old then old:Destroy() end

		local screenGui = Instance.new("ScreenGui")
		screenGui.Name = "MobileFluentToggle"
		screenGui.Parent = CoreGui
		screenGui.ResetOnSpawn = false

		local button = Instance.new("TextButton")
		button.Name = "ToggleButton"
		button.Parent = screenGui
		button.Size = UDim2.new(0, 52, 0, 52)
		button.Position = UDim2.new(0.04, 0, 0.22, 0)
		button.BackgroundColor3 = Color3.fromRGB(18, 20, 26)
		button.BorderSizePixel = 0
		button.Text = "HUB"
		button.TextColor3 = Color3.fromRGB(240, 240, 250)
		button.TextSize = 14
		button.Font = Enum.Font.GothamBold
		button.Active = true

		local corner = Instance.new("UICorner", button)
		corner.CornerRadius = UDim.new(0, 12)

		local stroke = Instance.new("UIStroke", button)
		stroke.Thickness = 1.5
		stroke.Color = Color3.fromRGB(50, 56, 72)

		local dragging, dragInput, dragStart, startPos = false, nil, nil, nil

		button.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging, dragStart, startPos = true, input.Position, button.Position
				input.Changed:Connect(function()
					if input.UserInputState == Enum.UserInputState.End then dragging = false end
				end)
			end
		end)

		button.InputChanged:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
				dragInput = input
			end
		end)

		UserInputService.InputChanged:Connect(function(input)
			if input == dragInput and dragging then
				local delta = input.Position - dragStart
				button.Position = UDim2.new(
					startPos.X.Scale, startPos.X.Offset + delta.X,
					startPos.Y.Scale, startPos.Y.Offset + delta.Y
				)
			end
		end)

		local isVisible = true
		button.MouseButton1Click:Connect(function()
			if dragging then return end
			isVisible = not isVisible
			if Window and Window.Root then Window.Root.Visible = isVisible end
		end)
	end)
end)

	Duration = 4
