local function humanDelay(minWait, maxWait)
	if not Config.HumanizeEnabled then task.wait(minWait or 0.1) return end
	local rMin = math.floor((minWait or Config.MinPreDelay) * 1000)
	local rMax = math.floor((maxWait or Config.MaxPreDelay) * 1000)
	if rMax <= rMin then rMax = rMin + 50 end
	task.wait(math.random(rMin, rMax) / 1000)
end

-- Detects mount and seat part
local function getActivePetMount()
	local char = localPlayer.Character
	if not char then return nil, nil end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum and hum.SeatPart then
		local model = hum.SeatPart:FindFirstAncestorOfClass("Model")
		if model and model ~= char then
			return model, hum.SeatPart
		end
	end
	return nil, nil
end

-- UNIFIED ASSEMBLY WARP: Moves pet assembly root to eliminate weld flinging/void
local function safeMountTeleport(targetCFrame)
	local char, hrp, hum = getCharacterParts()
	if not char or not hrp then return end

	-- Zero velocity across character
	for _, p in ipairs(char:GetDescendants()) do
		if p:IsA("BasePart") then
			p.AssemblyLinearVelocity = Vector3.zero
			p.AssemblyAngularVelocity = Vector3.zero
		end
	end

	local petModel, seatPart = getActivePetMount()
	if seatPart then
		-- Zero velocity on mount
		for _, p in ipairs(petModel:GetDescendants()) do
			if p:IsA("BasePart") then
				p.AssemblyLinearVelocity = Vector3.zero
				p.AssemblyAngularVelocity = Vector3.zero
			end
		end

		-- Teleporting the AssemblyRootPart moves both the mount and the seated rider together
		local rootAssembly = seatPart.AssemblyRootPart or seatPart
		rootAssembly.CFrame = targetCFrame
	else
		char:PivotTo(targetCFrame)
	end

	task.wait(0.12)
end

local function getObjectPosition(obj)
	if not obj or not obj.Parent then return nil end
	if obj:IsA("Model") then
		if obj.PrimaryPart then return obj.PrimaryPart.Position end
		local ok, cf = pcall(function() return obj:GetPivot() end)
		if ok and cf then return cf.Position end
	elseif obj:IsA("BasePart") then
		return obj.Position
	end
	return nil
end

local function autoDetectMyPlot()
	local myName, myUserId, myDisplayName = string.lower(localPlayer.Name), tostring(localPlayer.UserId), string.lower(localPlayer.DisplayName)
	local function isOwnerMatch(val)
		if not val then return false end
		if val == localPlayer then return true end
		local s = string.lower(tostring(val))
		return s == myName or s == myUserId or s == myDisplayName
	end

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Model") or obj:IsA("Folder") then
			local ownerVal = obj:FindFirstChild("Owner") or obj:FindFirstChild("UserId")
			local attrOwner = obj:GetAttribute("Owner") or obj:GetAttribute("UserId")
			if (ownerVal and isOwnerMatch(ownerVal.Value)) or (attrOwner and isOwnerMatch(attrOwner)) then
				local ok, cf = pcall(function() return obj:GetPivot() end)
				if ok and cf then
					State.HomePlotPosition = cf.Position
					return State.HomePlotPosition
				end
			end
		end
	end

	local _, hrp = getCharacterParts()
	State.HomePlotPosition = hrp and hrp.Position or Vector3.new(0, 10, 0)
	return State.HomePlotPosition
end

local function getMyPlotPosition()
	return State.HomePlotPosition or autoDetectMyPlot()
end

local function isEggBlacklisted(eggObj)
	local expireTime = State.FailedEggCache[eggObj]
	if not expireTime then return false end
	if os.clock() < expireTime then return true end
	State.FailedEggCache[eggObj] = nil
	return false
end

local function setNoClip(enable)
	if enable then
		if State.Connections.NoClip then return end
		local stepEvent = RunService.PreSimulation or RunService.Stepped

		State.Connections.NoClip = stepEvent:Connect(function()
			local char, hrp = getCharacterParts()
			if not char or not hrp then return end

			for _, part in ipairs(char:GetDescendants()) do
				if part:IsA("BasePart") then
					part.CanCollide = false
				end
			end

			local petModel = getActivePetMount()
			if petModel then
				for _, part in ipairs(petModel:GetDescendants()) do
					if part:IsA("BasePart") then
						part.CanCollide = false
					end
				end
			end
		end)
	else
		if State.Connections.NoClip then
			State.Connections.NoClip:Disconnect()
			State.Connections.NoClip = nil
		end

		local char, hrp = getCharacterParts()
		if char then
			for _, part in ipairs(char:GetDescendants()) do
				if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
					part.CanCollide = true
				end
			end
		end
		if hrp then
			hrp.AssemblyLinearVelocity = Vector3.zero
		end
	end
end

local function disableScreenShake(enable)
	if enable then
		if State.Connections.ScreenShake then return end
		local function muteShakes(parent)
			if not parent then return end
			for _, scriptObj in ipairs(parent:GetDescendants()) do
				if scriptObj:IsA("LocalScript") then
					local n = string.lower(scriptObj.Name)
					if string.find(n, "shake") or string.find(n, "recoil") or string.find(n, "camoffset") then
						pcall(function() scriptObj.Disabled = true end)
					end
				end
			end
		end

		muteShakes(localPlayer:FindFirstChild("PlayerScripts"))
		muteShakes(localPlayer.Character)

		State.Connections.ScreenShake = RunService.RenderStepped:Connect(function()
			local char = localPlayer.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if hum then hum.CameraOffset = Vector3.zero end
		end)
	else
		if State.Connections.ScreenShake then
			State.Connections.ScreenShake:Disconnect()
			State.Connections.ScreenShake = nil
		end
	end
end

local function executeRawTween(targetCFrame, speed)
	local char, hrp = getCharacterParts()
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum or hum.Health <= 0 then return nil, 0 end

	if State.CurrentTween then
		pcall(function() State.CurrentTween:Cancel() end)
		State.CurrentTween = nil
	end

	local distance = (targetCFrame.Position - hrp.Position).Magnitude
	local safeSpeed = math.max(15, speed or Config.OutboundSpeed)
	local duration = math.max(distance / safeSpeed, 0.05)

	hrp.AssemblyLinearVelocity = Vector3.zero
	hrp.AssemblyAngularVelocity = Vector3.zero
	if Config.AnchorDuringFlight then hrp.Anchored = true end

	local _, seatPart = getActivePetMount()
	if seatPart and Config.AnchorDuringFlight then pcall(function() seatPart.Anchored = true end) end

	local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
	State.CurrentTween = TweenService:Create(hrp, tweenInfo, {CFrame = targetCFrame})
	State.CurrentTween:Play()
	return State.CurrentTween, duration
end

local function safeWaitTween(tween, expectedDuration)
	if not tween then return false end
	local finished = false
	local conn = tween.Completed:Connect(function() finished = true end)
	local startTime = os.clock()

	while not finished and (os.clock() - startTime < (expectedDuration or 5) + 3) do
		if not State.IsRunning and not State.IsCandyRunning then
			pcall(function() tween:Cancel() end)
			if conn then conn:Disconnect() end
			local _, hrp = getCharacterParts()
			if hrp then hrp.Anchored = false end
			return false
		end
		task.wait(0.04)
	end
	if conn then conn:Disconnect() end
	return finished
end

local function smartMoveTo(targetPosition, customSpeed, landOffset)
	local destPos = targetPosition + (landOffset or Vector3.new(0, 1.5, 0))
	local tw, expectedTime = executeRawTween(CFrame.new(destPos), customSpeed or Config.OutboundSpeed)
	local success = safeWaitTween(tw, expectedTime)

	local _, hrp = getCharacterParts()
	if hrp then
		hrp.Anchored = false
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
	end
	return success
end

local function autoAlignViewAtTarget(targetPos)
	if not Config.AutoCameraAlign then return end
	local char = localPlayer.Character
	if char then
		local pet = getActivePetMount()
		local root = pet or char
		root:PivotTo(CFrame.lookAt(root:GetPivot().Position, Vector3.new(targetPos.X, root:GetPivot().Position.Y, targetPos.Z)))
	end
end

local function getCandyTokenValue(object)
	if not object then return 1 end

	local amountGui = object:FindFirstChild("CandyAmount", true)
	if amountGui then
		local label = amountGui:FindFirstChild("Amount", true)
		if label and label.Text ~= "" then
			local num = tonumber(string.match(label.Text, "%d+"))
			if num then return num end
		end
	end

	local name = string.lower(object.Name)
	if string.find(name, "candy_04") or string.find(name, "mythic") then
		return 5
	elseif string.find(name, "candy_03") or string.find(name, "rare") then
		return 3
	elseif string.find(name, "candy_02") then
		return 2
	end
	return 1
end

local function exactSpawnIdResolver(visualMesh)
	local id = visualMesh:GetAttribute("SpawnId") or visualMesh:GetAttribute("Id")
	if id and type(id) == "string" and id ~= "" then return id end

	for _, spawnPart in ipairs(CollectionService:GetTagged("HalloweenCandySpawn")) do
		if spawnPart:IsA("BasePart") and (spawnPart.Position - visualMesh.Position).Magnitude <= 8 then
			return spawnPart:GetAttribute("Id") or spawnPart:GetFullName()
		end
	end
	return visualMesh.Name
end

local function getAllCandyCandidates()
	local list = {}
	local seen = {}

	local hwFolder = workspace:FindFirstChild("HalloweenCandy")
	if hwFolder then
		for _, visual in ipairs(hwFolder:GetChildren()) do
			if visual:IsA("BasePart") and visual.Transparency < 1 and not seen[visual] and not isEggBlacklisted(visual) then
				seen[visual] = true
				table.insert(list, {
					Object = visual,
					Position = visual.Position,
					SpawnId = exactSpawnIdResolver(visual),
					Tokens = getCandyTokenValue(visual)
				})
			end
		end
	end

	table.sort(list, function(a, b) return a.Tokens > b.Tokens end)
	return list
end

local function executeInstantDeposit(candyOriginPos, pickupTime)
	local _, dropPos = getCandyDropOffPoint()
	if not dropPos then return false end

	-- Anti-Cheat Math: Calculate exact cooldown so Speed <= 75 studs/s
	local distance = candyOriginPos and (dropPos - candyOriginPos).Magnitude or 180
	local minTimeAllowed = distance / 70 -- Time server expects you to take
	local elapsed = os.clock() - (pickupTime or os.clock())
	local waitRemaining = math.max(Config.DeliveryDwellTime, minTimeAllowed - elapsed)

	-- 1. Dwell at candy for the required 3 seconds (server speed validation passes!)
	updateCandyStatus(string.format("Anti-Cheat delay (%.1fs dwell)...", waitRemaining))
	task.wait(waitRemaining)

	-- 2. INSTANT TELEPORT directly into the CandyDropOff square box
	updateCandyStatus("Warping into CandyDropOff pad...")
	safeMountTeleport(CFrame.new(dropPos + Vector3.new(0, 1.5, 0)))

	-- 3. Stand on pad and wait for automatic collection
	local depositDeadline = os.clock() + 2.0
	local confirmed = false

	while (os.clock() < depositDeadline) and State.IsCandyRunning do
		task.wait(0.2)
		if getCarriedCandyCount() == 0 then
			confirmed = true
			break
		end
	end

	if confirmed then
		State.CandiesStolenCount = State.CandiesStolenCount + 1
		updateCandyStatus("Token Awarded! (" .. State.CandiesStolenCount .. " total)")
		return true
	end

	return false
end

local function attemptVerifiedPickup(candyEntry)
	local remotes = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("Game")
	local collectRemote = remotes and remotes:FindFirstChild("CollectCandy")

	local initialCount = getCarriedCandyCount()
	local startTime = os.clock()
	local secured = false

	autoAlignViewAtTarget(candyEntry.Position)

	while (os.clock() - startTime < 1.6) and State.IsCandyRunning do
		if collectRemote and candyEntry.SpawnId then
			collectRemote:FireServer(candyEntry.SpawnId)
		end

		pcall(function()
			for _, p in ipairs(candyEntry.Object:GetDescendants()) do
				if p:IsA("ProximityPrompt") then
					p.Enabled = true
					p.RequiresLineOfSight = false
					p.HoldDuration = 0
					p.MaxActivationDistance = 999
					p:SetAttribute("CandyPromptAvailable", true)
					if fireproximityprompt then fireproximityprompt(p) end
				end
			end
		end)

		task.wait(0.2)

		if getCarriedCandyCount() > initialCount then
			secured = true
			break
		end
	end

	return secured
end

local function executeEggInteraction(eggObj)
	if not State.IsRunning or not eggObj or not eggObj.Parent then return false end
	local _, hrp = getCharacterParts()
	if not hrp then return false end

	humanDelay(Config.MinPreDelay, Config.MaxPreDelay)
	local interacted = false

	pcall(function()
		for _, prompt in ipairs(eggObj:GetDescendants()) do
			if prompt:IsA("ProximityPrompt") then
				prompt.RequiresLineOfSight = false
				prompt.HoldDuration = 0
				prompt.MaxActivationDistance = 999
				prompt.Enabled = true
				if fireproximityprompt then
					fireproximityprompt(prompt)
					interacted = true
				end
			end
		end
	end)

	pcall(function()
		local touchTarget = eggObj:IsA("BasePart") and eggObj or eggObj:FindFirstChildWhichIsA("BasePart", true)
		if touchTarget and firetouchinterest then
			firetouchinterest(hrp, touchTarget, 0)
			task.wait(0.02)
			firetouchinterest(hrp, touchTarget, 1)
			interacted = true
		end
	end)

	humanDelay(Config.WaitAtEgg, Config.WaitAtEgg + 0.1)
	return interacted
end

local function getEggRarity(obj)
	if not obj then return "unknown" end
	local attr = obj:GetAttribute("Rarity") or obj:GetAttribute("EggRarity") or obj:GetAttribute("Tier")
	if attr and type(attr) == "string" and attr ~= "" then return string.lower(attr) end
	local valObj = obj:FindFirstChild("Rarity") or obj:FindFirstChild("EggRarity") or obj:FindFirstChild("Tier")
	if valObj and valObj:IsA("StringValue") and valObj.Value ~= "" then return string.lower(valObj.Value) end

	local name = string.lower(obj.Name)
	if string.find(name, "blackhole") or string.find(name, "solaris") or string.find(name, "cherub")
		or string.find(name, "volcanic") or string.find(name, "ethereal") then return "ethereal" end
	if string.find(name, "aurora") or string.find(name, "galaxy") or string.find(name, "bloom") or string.find(name, "divine") then return "divine" end
	if string.find(name, "diamond") or string.find(name, "crystal") or string.find(name, "skull")
		or string.find(name, "asteroid") or string.find(name, "dominus") or string.find(name, "flaming")
		or string.find(name, "sinister") or string.find(name, "soul") or string.find(name, "tidal")
		or string.find(name, "mythic") then return "mythic" end
	if string.find(name, "glass") or string.find(name, "golden") or string.find(name, "gold") or string.find(name, "legendary") then return "legendary" end
	if string.find(name, "mushroom") or string.find(name, "flower") or string.find(name, "slime") or string.find(name, "ice") or string.find(name, "epic") then return "epic" end
	if string.find(name, "cracked") or string.find(name, "easter") or string.find(name, "stone") or string.find(name, "leaf") or string.find(name, "rare") then return "rare" end
	if string.find(name, "white") or string.find(name, "brown") or string.find(name, "common") then return "common" end
	return "unknown"
end

local function getRootEggObject(obj)
	if not obj then return nil end
	if obj.Parent and obj.Parent ~= workspace and obj.Parent:IsA("Model") then
		local pName = string.lower(obj.Parent.Name)
		if string.find(pName, "egg") or getEggRarity(obj.Parent) ~= "unknown" then
			return obj.Parent
		end
	end
	return obj
end

local function getActiveEggObjects()
	local targetEggs = {}
	local processedSet = {}
	local targetRarity = string.lower(Config.TargetRarity)

	local function processCandidate(candidate)
		if not candidate or not candidate.Parent then return end
		local eggObj = getRootEggObject(candidate)
		if processedSet[eggObj] then return end

		if Players:GetPlayerFromCharacter(eggObj) or (localPlayer.Character and eggObj:IsDescendantOf(localPlayer.Character)) then return end
		if isEggBlacklisted(eggObj) then return end

		local eggName = string.lower(eggObj.Name)
		local detectedRarity = getEggRarity(eggObj)

		local isAnEgg = string.find(eggName, "egg") or (detectedRarity ~= "unknown")
		if not isAnEgg then return end

		local matches = (targetRarity == "all") or (detectedRarity == targetRarity)
		if matches then
			processedSet[eggObj] = true
			table.insert(targetEggs, eggObj)
		end
	end

	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Model") or obj:IsA("BasePart") then
			processCandidate(obj)
		end
	end

	return targetEggs
end

local function applyPerformanceBoost(enable)
	if not enable then return end
	task.spawn(function()
		pcall(function()
			settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
			local terrain = workspace:FindFirstChildOfClass("Terrain")
			if terrain then
				terrain.WaterWaveSize = 0
				terrain.WaterWaveSpeed = 0
			end
			for _, obj in ipairs(workspace:GetDescendants()) do
				if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
					obj.Enabled = false
				elseif obj:IsA("Beam") or obj:IsA("Highlight") then
					obj.Enabled = false
				end
			end
		end)
	end)
end

local function cleanupAll()
	State.IsRunning = false
	State.IsCandyRunning = false
	if State.CurrentTween then
		pcall(function() State.CurrentTween:Cancel() end)
		State.CurrentTween = nil
	end
	setNoClip(false)
	disableScreenShake(false)

	local _, hrp = getCharacterParts()
	local _, seat = getActivePetMount()
	if hrp then
		hrp.Anchored = false
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
	end
	if seat then pcall(function() seat.Anchored = false end) end

	updateStatus("Idle")
	updateCandyStatus("Idle")
	task.defer(function() pcall(function() collectgarbage("step", 50) end) end)
end

local function serverHop()
	local placeId = game.PlaceId
	local curJob = game.JobId
	local url = "https://games.roblox.com/v1/games/" .. placeId .. "/servers/Public?sortOrder=Asc&limit=100"

	local success, result = pcall(function()
		return HttpService:JSONDecode(game:HttpGet(url))
	end)

	if success and result and result.data then
		for _, server in ipairs(result.data) do
			if server.playing < server.maxPlayers and server.id ~= curJob then
				TeleportService:TeleportToPlaceInstance(placeId, server.id, localPlayer)
				return
			end
		end
	end
	TeleportService:Teleport(placeId, localPlayer)
end

					startPos.Y.Scale, startPos.Y.Offset + delta.Y
